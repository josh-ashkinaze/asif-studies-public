  # ── SCENARIO DESCRIPTION ─────────────────────
  # Passed to every agent as the event name.
  # Agents generate 4 quantities from this +
  # the participant's intake answers:
  #   start_of_event          (~100-130 words)
  #   event_effects           (effects list)
  #   event_plot              (~50-word plot)
  #   event_plot_from_effects (~100-word plot)
  name: >
    Gravity has suddenly stopped working.
    Everything you normally do today — your
    routines, your commute, your meals — now
    happens in zero gravity. Walk through your
    day from the moment you wake up.

  # ── WHAT FEEDS THE AI AUTOCOMPLETER ──────────
  # Selects which generated quantity is passed
  # as event context to the completion LLM.
  #  "start"             → start_of_event
  #  "plot"              → event_plot
  #  "plot_from_effects" → event_plot_from_effects
  #  "effects"           → event_effects
  #  "none"              → no event context
  autocomplete_context: "plot_from_effects"

  # ── WHAT THE PARTICIPANT SEES ─────────────────
  # Selects which quantity is shown above the
  # editor to set the scene. Can differ from
  # autocomplete_context — e.g. show participant
  # the opening sentence while feeding the LLM
  # the richer plot. Same options as above.
  shown_to_user: "start"
  shown_to_user_prefix: "Imagine this happens
    during your regular day"

  # ── EDITOR PLACEHOLDER ────────────────────────
  # Gray hint text shown before the participant

  editor_placeholder: >

    and reached to turn off my alarm clock.
    As soon as I touched it, I realized...
    [writing assistant will autocomplete]...'
